Uses 100 hard-coded “random-like” int16 values in the range ≈ ±32768

Computes:

value * 0.061f

value * 0.06103515625f (exact 2g scale = 2000 mg / 32768)

Measures execution time for each method using micros()

Computes average absolute difference between the two results

Prints everything clearly




#include <Arduino.h>

#define N 100

// 100 hardcoded "random-like" int16 values (~ ±32768)
int16_t rawVals[N] = {
  -32760, -30000, -28000, -25000, -22000, -20000, -18000, -15000, -12000, -10000,
   -9000,  -8000,  -7000,  -6000,  -5000,  -4000,  -3000,  -2000,  -1000,   -500,
       0,    500,   1000,   2000,   3000,   4000,   5000,   6000,   7000,   8000,
    9000,  10000,  12000,  15000,  18000,  20000,  22000,  25000,  28000,  30000,
  -31000, -29000, -27000, -24000, -21000, -19000, -17000, -14000, -11000,  -9500,
   -8500,  -7500,  -6500,  -5500,  -4500,  -3500,  -2500,  -1500,   -750,   -250,
     250,    750,   1500,   2500,   3500,   4500,   5500,   6500,   7500,   8500,
    9500,  11000,  14000,  17000,  19000,  21000,  24000,  27000,  29000,  31000,
  -32700, -16384, -8192, -4096, -2048, -1024, 1024, 2048, 4096, 8192
};

float outA[N];
float outB[N];

void setup() {
  Serial.begin(115200);
  while (!Serial);

  Serial.println("=== Precision & Timing Test ===");

  // ---------------- Method A : 0.061 ----------------
  unsigned long t1 = micros();
  for (int i = 0; i < N; i++) {
    outA[i] = rawVals[i] * 0.061f;
  }
  unsigned long t2 = micros();

  // ---------------- Method B : 0.06103515625 ----------------
  unsigned long t3 = micros();
  for (int i = 0; i < N; i++) {
    outB[i] = rawVals[i] * 0.06103515625f;
  }
  unsigned long t4 = micros();

  // ---------------- Difference calculation ----------------
  double sumDiff = 0.0;
  double maxDiff = 0.0;

  for (int i = 0; i < N; i++) {
    double diff = fabs(outA[i] - outB[i]);
    sumDiff += diff;
    if (diff > maxDiff) maxDiff = diff;
  }

  double avgDiff = sumDiff / N;

  // ---------------- Results ----------------
  Serial.print("Time using 0.061f            : ");
  Serial.print(t2 - t1);
  Serial.println(" us");

  Serial.print("Time using 0.06103515625f    : ");
  Serial.print(t4 - t3);
  Serial.println(" us");

  Serial.println();
  Serial.print("Average difference (mg)      : ");
  Serial.println(avgDiff, 10);

  Serial.print("Maximum difference (mg)      : ");
  Serial.println(maxDiff, 10);

  Serial.println("\nSample values:");
  for (int i = 0; i < 5; i++) {
    Serial.print("Raw: ");
    Serial.print(rawVals[i]);
    Serial.print("  0.061: ");
    Serial.print(outA[i], 10);
    Serial.print("  Exact: ");
    Serial.println(outB[i], 10);
  }
}
